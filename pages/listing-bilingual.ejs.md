<%
// Bilingual listing template.
//
// A custom `template:` only receives `items` (already sorted/filtered by
// Quarto core) and `utils` (b64encode/b64decode) — NOT `listing` or its
// utilities. So we hand-roll the `.quarto-post` markup that quarto-listing.js
// expects (data-index, data-categories, data-listing-*-sort).
//
// For each card we pick the `<field>-en` value when the active profile renders
// in English (item.lang starts with "en"), falling back to the canonical field.
// Localized: title, subtitle, description.
// Categories keep their canonical (frontmatter `categories`) values so the
// category sidebar — generated by Quarto core, always canonical — and
// click-to-filter stay in sync.
%>

::: {.list .quarto-listing-default}

<% items.forEach(function(item, index) { %>
<%
const en = item.lang && String(item.lang).startsWith("en");
const pick = (f) => (en && item[f + "-en"] !== undefined && item[f + "-en"] !== null) ? item[f + "-en"] : item[f];
const itemTitle = pick("title");
const itemSubtitle = pick("subtitle");
const itemDescription = pick("description");

const sv = item.sortableValues || {};
let attrs = `data-index="${index}"`;
if (item.categories && item.categories.length) {
  attrs += ` data-categories="${utils.b64encode(item.categories.join(','))}"`;
}
Object.keys(sv).forEach(function(k) {
  attrs += ` data-listing-${k}-sort="${sv[k]}"`;
});
const align = item['image-align'] || 'left';
%>

```{=html}
<div class="quarto-post image-<%= align %>" <%= attrs %>>
```

<% if (item.image) { %>

```{=html}
<div class="thumbnail"><a href="<%- item.path %>" class="no-external"><img src="<%- item.image %>" class="thumbnail-image" alt="<%= item['image-alt'] || '' %>"/></a></div>
```

<% } %>

```{=html}
<div class="body">
```

<h3 class="no-anchor listing-title"><a href="<%- item.path %>" class="no-external"><%= itemTitle %></a></h3>

<% if (itemSubtitle) { %>
<div class="listing-subtitle"><a href="<%- item.path %>" class="no-external"><%= itemSubtitle %></a></div>
<% } %>

<% if (item.categories && item.categories.length) { %>

```{=html}
<div class="listing-categories">
<% for (const category of item.categories) { %>
<div class="listing-category" onclick="window.quartoListingCategory('<%= utils.b64encode(category) %>'); return false;"><%= category %></div>
<% } %>
</div>
```

<% } %>

<% if (itemDescription) { %>

```{=html}
<div class="delink listing-description"><a href="<%- item.path %>" class="no-external">
```

<%= itemDescription %>

```{=html}
</a></div>
```

<% } %>

```{=html}
</div>
<div class="metadata">
<a href="<%- item.path %>" class="no-external">
```

<% if (item.date) { %>
<div class="listing-date"><%= item.date %></div>
<% } %>

<% if (item.author) { %>
<div class="listing-author"><%= item.author %></div>
<% } %>

```{=html}
</a>
</div>
</div>
```

<% }); %>

:::
